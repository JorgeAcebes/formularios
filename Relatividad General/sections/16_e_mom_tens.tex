\section*{Tensor Energía-Momento}
\begin{align*}
    &\text{Densidad de corriente: } \underline{J} =  \rho \,c\, e_0+ \rho \, v^ie_i  \xrightarrow[\text{reposo}]{\text{observador en }} \, =\rho_o \,c\, e_{0'}\\
    &  \rho\equiv \text{densidad de carga}\; \; \; \; \vec{v}\equiv \text{velocidad cargas}\; \; \; \; \rho=\gamma  \rho_0\\
    &\text{Divergencia de un cuadrivector: }\text{div}\underline{A}=\text{tr}(\nabla_\alpha A^\beta) = \nabla_\alpha A^\alpha \\
    &\text{Ecuación de continuidad (de manera general): }\nabla_\mu J^\mu =0\\
    &\text{Ec. de continuidad en cartesinanas: }\partial_\mu J^\mu = \frac{\partial J^\mu}{\partial x^\mu}=0 \\
    &\text{Ec. de continuidad en cilíndricas: } \partial_0 (\rho c) \!+\! \partial_r (\rho v^r ) \!+\! \partial_\phi (\rho v^\phi)+ \frac{1}{r}\rho v^r \!+\! \partial_z (\rho v^z)\!=\!0\\
    &\text{S.R. reposo: densidad de partículas: }n_0 = \frac{N}{V_0}\; \; \; \;
    \text{ densidad de masa: $\rho_0=n_0\, m$}\\
    &\text{densidad de energía: $\varepsilon_0=\rho_0\, c^2$}\; \; \; \;  
    \text{densidad de momento: } \pi_0=n_0\, p_0=\varepsilon_0 \, \frac{v}{c^2}=0\\
    &\text{Según S.R. inercial: $n\!=\!n_0\, \gamma \; \; \; \; \rho\!=\!\rho_0\, \gamma\!=\!n\,m\;\; \; \; \varepsilon\!=\!\varepsilon_0\,\gamma^2\!=\rho\, \gamma \, c^2 \; \; \; \; \pi\!=\!n\, p\!=\!\varepsilon\, \frac{v}{c^2} $}\\
    &\text{Tensor energía-momento: }\mathcal{T}\!=\!\rho_0 \, \underline{u}\!\otimes \!\underline{u}\;\; \;\; \text{Conservación energía-momento: }\nabla_\mu T^{\mu\nu}\!=\!0
\end{align*}
