\section*{Gravedad según Newton \normalfont{$(\dot{r}= \frac{dr}{dt}, \; \dot{\phi}= \frac{d\phi}{dt})$}}
\begin{align*}
    &\text{Energía (conservada): } E= \frac{1}{2}mv^2 - \frac{GMm}{r}\Rightarrow E = \frac{1}{2}m(\dot{r}^2+ r^2 \dot{\phi}^2 )- \frac{GMm}{r}\\
    &\text{Momento angular (conservado): } L=L_z = mr^2 \dot{\phi}\Rightarrow\text{ trayectorias en un plano}\\
    &E=\frac{1}{2}m\dot{r}^2 +V_{\text{eff}}(r)\;  \Rightarrow \;\text{Potencial efectivo: } V_{\text{eff}}(r)= \frac{L^2}{2mr^2}- \frac{GMm}{r}\\
    &\text{Ecuación diferencial movimiento: } \frac{d^2 u}{d\phi^2} + u=\frac{GMm^2}{L^2}\; \; \;u \equiv \frac{1}{r}\\
    &\text{Solución analítica: }r(\phi)=\frac{L^2}{GMm^2(1+\varepsilon \cos \phi)}\; \; \; \; \; \varepsilon \equiv \frac{K}{GMm/L^2}\equiv\text{cte}\\
    &\varepsilon = 0 \rightarrow \text{circunferencia}, \; \; \varepsilon < 1 \rightarrow \text{elipse}, \; \; 
    \varepsilon = 1 \rightarrow \text{parábola}, \; \; 
    \varepsilon >1 \rightarrow \text{hipérbola}
\end{align*}
