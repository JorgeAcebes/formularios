\section*{Observadores estáticos}
\begin{align*}
    &\text{Observador: definido por base ortonormal  de cuadrivectores } \{\alpha_0, \alpha_1, \alpha_2, \alpha_3\}\\
    &\alpha_0\cdot\alpha_0 =-1 \; \; \;\; \alpha_1\cdot\alpha_1 =\alpha_2\cdot\alpha_2 =\alpha_3\cdot\alpha_3 =1\; \; \; \;\alpha_i\cdot\alpha_j = 0 \; \forall \; i\neq j\\
    &\text{$\{e_0,e_1, e_2,e_3\}_{\scalebox{0.8}{\text{Schwarzschild}}}$ solo corresponde a un observador (lejano) cuando } r\rightarrow \infty\\   \
    &\tilde{e}_0\!=\!\frac{1}{\sqrt{1-\frac{1}{r}}}\,e_0\;\;\; \tilde{e}_1\!=\! \sqrt{1-\frac{1}{r}}\,e_1\;\;\; \tilde{e}_2\! =\! \frac{1}{r} \, e_2 \; \; \; \tilde{e}_3\!=\! \frac{1}{r\sin\theta}\,e_3  \!\!\!\!\!\!\!\!\!\!
    \begin{array}{cc}
         &\{\tilde{e}_0,\tilde{e}_1,\tilde{e}_2,\tilde{e}_3\}_{\scalebox{0.7}{\text{Schwarzs.}}} \vspace{3pt} \\
         & \text{es observador}
    \end{array}
    \\
    &\text{Observador estático: $r, \theta, \phi \equiv $ ctes $\forall \; t \Rightarrow$ NO sigue geodésica $\Rightarrow$ necesita aceleración}\\
    &\text{Espacio-tiempo estático: aquel donde existen coordenadas en las que los coeficientes}\\
    &\text{de la métrica no dependen del tiempo $\Leftrightarrow \exists\; \;  \underline{\xi}^{\text{tiempo}}$}\\
    &\text{Observador estático en Schwarzschild: } \underline{u}=\tilde{e}_0\; \; \; \; \;\; \underline{a}=\frac{1}{2r^2}\frac{1}{\sqrt{1-\frac{1}{r}}}\,\tilde{e}_1 \\
    &\text{En S.I.: } a= \frac{GM}{r^2} \frac{1}{\sqrt{1-\frac{2GM}{c^2r}}} \;\;\;\;\;\; \text{Según Newton: } a=\frac{GM}{r^2}
\end{align*}
