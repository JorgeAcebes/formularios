\section*{Tensor de Riemman $\mathcal{R}=R^\alpha_{\;\;\beta\gamma\delta} \;e_\alpha\otimes e^\beta \otimes e^\gamma \otimes e^\delta\; \;\; \mathcal{R}=0\Leftrightarrow \normalfont\text{espacio-tiempo plano}$} 
\begin{align*}
&R^\alpha_{\; \; \beta \gamma \delta } = \frac{\partial \Gamma^\alpha_{\delta \beta }}{\partial  x^\gamma}- 
\frac{\partial \Gamma^\alpha_{\gamma\beta}}{\partial x^\delta}+\Gamma^\alpha_{\gamma\sigma}\Gamma^\sigma_{\delta\beta}-\Gamma^\alpha_{\delta\sigma}\Gamma^\sigma_{\gamma\beta}\; \; \; 
R_{\alpha \beta \gamma \delta }= g_{\alpha \sigma}R^\sigma_{\;\;\beta \gamma\delta}\\&
\partial_\alpha \partial_\beta A^\mu e_\mu - \partial_\beta \partial_\alpha A^\mu e_\mu = R^\nu_{\;\;\gamma\alpha\beta}A^\gamma e_\nu\;\;\; \left(\partial_\alpha\partial_\beta-\partial_\beta\partial \alpha \right)e_\gamma = R^\sigma_{\;\; \gamma\alpha\beta}e_\sigma
\\&\nabla_\sigma R^\alpha_{\;\;\beta \gamma \delta}= \partial_\sigma R^\alpha _{\;\;\beta \gamma\delta}+\Gamma^\alpha_{\omega\sigma}R^\omega_{\;\;\beta \gamma \delta }-\Gamma^\rho_{\beta\sigma}R^\alpha_{\;\;\rho\gamma\delta}-\Gamma^\phi_{\gamma\sigma}R^\alpha_{\;\;\beta\phi\delta}-
\Gamma^\theta_{\delta\sigma}R^\alpha_{\;\;\beta\gamma\theta}\\
&\text{Tensor de Ricci: } R^\beta_{\;\;\alpha\beta\gamma} \equiv
R_{\alpha\gamma}\;\;\; \text{Curvatura escalar: } R=g^{\alpha \beta}R_{\alpha \beta}
\end{align*}
\underline{Propiedades}:
\begin{align*}
&R_{\alpha \beta \gamma\delta}{=}- R_{\beta\alpha \gamma\delta}\;\;\;
R_{\alpha\beta\gamma\delta}{=}- R_{\alpha \beta \delta \gamma} \;\;\; R_{\alpha\beta\gamma\delta}{=} R_{\gamma \delta \alpha\beta} \;\;\; R_{\alpha \beta}{=}R_{\beta\alpha}\;\;\; R^{\alpha\beta}{=}R^{\beta\alpha}  \\ 
&R_{\alpha \beta\delta \gamma} + R_{\alpha \delta \beta \gamma}+R_{\alpha\gamma\delta\beta}=0  \;\;\;\partial_\beta R^{\alpha\beta}=\frac{1}{2}g^{\alpha \beta}\partial_\beta R 
\\
& \text{Identidad de Bianchi: }\partial_\sigma R^\alpha_{\;\;\beta\rho\omega}+ \partial_\omega R^\alpha_{\;\;\beta\sigma\rho}+
\partial_\rho R^\alpha_{\;\;\beta\omega\sigma} =0\\&
R_{\alpha \beta}=\frac{1}{\sqrt{|g|}}\partial_\gamma (\sqrt{|g|}\Gamma^\gamma_{\alpha \beta} )-\partial_\alpha\partial_\beta(\ln\sqrt{|g|})-\Gamma^\gamma_{\delta\alpha}\Gamma^\delta_{\beta\gamma}\\
&\text{Número componentes independientes de }\mathcal{R}: \frac{n^2}{12}(n^2-1)\;\;\; n\equiv\text{nº dimensiones}
\end{align*}
