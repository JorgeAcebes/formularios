\section*{Ecuaciones de Campo de Einstein y Constante Cosmológica}
\begin{align*}
    &\text{Métrica de campo gravitatorio uniforme y débil ($V/c \ll 1$): }\\&\quad\quad \quad g_{00} = -1-\frac{2V}{c^2}, \; g_{11} =g_{22} =
    g_{33} = 1, \text{ resto = 0} \\
    & \text{Símbolos de Christoffel: } \Gamma^{\alpha}_{00} = \frac{1}{c^2}g^{\alpha \mu} \partial_{\mu} V\;\;\; \Gamma^{0}_{\beta 0} = \frac{1}{c^2} \frac{1}{1+\frac{2V}{c^2}} \partial_{\beta}V, \;\; \text{resto} = 0.  
    \\& \text{Ecuación de Campo de Einstein: } R_{\alpha \beta}- \frac{1}{2} g_{\alpha \beta }R = \frac{8\pi G}{c^4}T_{\alpha \beta} 
    \\& \text{Acción: } S[g] = -\frac{c^4}{16\pi G} \int d^4 x \sqrt{-g} R - \frac{c^4}{8\pi G} \int d^3 p \sqrt{|h|}K + \int d^4 x\sqrt{-g} \text{ L}_m\\&
    \text{Ecuación de Campo de Einstein con la constante cosmológica: }\\&
    \quad\quad \quad R_{\alpha \beta}- \frac{1}{2} g_{\alpha \beta }R + g_{\alpha \beta} \Gamma  = \frac{8\pi G}{c^4}T_{\alpha \beta}      
    \\& \text{Acción: } S[g] = -\frac{c^4}{16\pi G} \int d^4 x \sqrt{-g} (R-2\Gamma) - \frac{c^4}{8\pi G} \int d^3 p \sqrt{|h|}K + \int d^4 x\sqrt{-g} \text{ L}_m
    \end{align*}
